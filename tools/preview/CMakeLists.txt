# hagra-preview — QML 파일 하나를 HagraUI 모듈과 함께 띄우는 미리보기 실행기 (갤러리·대표 장면·디자인 검증용)
set(CMAKE_AUTOMOC ON)
qt_add_executable(hagra-preview main.cpp)
target_compile_definitions(hagra-preview PRIVATE HAGRA_QML_INSTALL_DIR="${HAGRA_QML_INSTALL_DIR}" HAGRA_VERSION="${PROJECT_VERSION}")
target_link_libraries(hagra-preview PRIVATE hagratheme Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
install(TARGETS hagra-preview RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
