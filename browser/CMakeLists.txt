# 하그라 브라우저 — hagra-browser (Qt6 WebEngine Quick + HagraUI + HagraTheme). Qt6WebEngineQuick 이 없으면 건너뛴다.
find_package(Qt6 6.4 COMPONENTS WebEngineQuick QUIET)
if(NOT Qt6WebEngineQuick_FOUND)
  message(WARNING "하그라: Qt6WebEngineQuick not found (qt6-webengine-dev) — hagra-browser skipped")
  return()
endif()
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

qt_add_executable(hagra-browser src/main.cpp src/browserhelper.cpp src/browserhelper.h qml/qml.qrc)
target_include_directories(hagra-browser PRIVATE src)
target_compile_definitions(hagra-browser PRIVATE HAGRA_QML_INSTALL_DIR="${HAGRA_QML_INSTALL_DIR}" HAGRA_VERSION="${PROJECT_VERSION}"
  HAGRA_TRANSLATIONS_DIR="${CMAKE_INSTALL_PREFIX}/${HAGRA_TRANSLATIONS_INSTALL_DIR}")
hagra_add_translations(hagra-browser ${CMAKE_CURRENT_SOURCE_DIR}/i18n/hagra-browser_en.ts)
target_link_libraries(hagra-browser PRIVATE hagratheme Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::DBus Qt6::Network Qt6::WebEngineQuick)
install(TARGETS hagra-browser RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES org.hagra.browser.desktop hagra-mimeapps.list DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES org.hagra.browser.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)

if(HAGRA_BUILD_TESTS)
  find_package(Qt6 COMPONENTS Test QUIET)
  if(Qt6Test_FOUND)
    qt_add_executable(tst_browser tests/tst_browser.cpp src/browserhelper.cpp src/browserhelper.h)
    target_include_directories(tst_browser PRIVATE src)
    target_link_libraries(tst_browser PRIVATE Qt6::Core Qt6::DBus Qt6::Test)
    add_test(NAME browser.helper COMMAND tst_browser)
    set_tests_properties(browser.helper PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
endif()
